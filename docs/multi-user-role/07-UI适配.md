# 多用户角色扩展方案 — UI 适配

## 1. Layout 导航（`src/views/layout.ejs`）

```ejs
<% if (user) { %>
  <a href="/">首页</a>
  <a href="/articles">文章</a>
  <a href="/history">总结</a>
  <a href="/search">搜索</a>
  <a href="/deepsearch">深度检索</a>
  <a href="/filter-logs">日志</a>
  <% if (user.role !== 'guest') { %>           <%-- 改：guest 隐藏，user/admin 可见 --%>
    <a href="/settings">设置</a>
    <a href="/topics">主题</a>
  <% } %>
  <form method="POST" action="/logout">
    <button type="submit" class="btn-icon">登出</button>
  </form>
<% } else { %>
  <a href="/login">登录</a>
<% } %>
```

## 2. 首页（`src/views/index.ejs`）

```javascript
// 客户端角色判断
if (window.userRole === 'guest') {
  document.querySelectorAll('[data-guest-hide]').forEach(el => el.style.display = 'none');
  document.querySelectorAll('.guest-only').forEach(el => el.style.display = 'block');
  document.querySelectorAll('.admin-message').forEach(el => el.style.display = 'none');
  document.querySelectorAll('.guest-message').forEach(el => el.style.display = 'block');
}
```

`data-guest-hide` 元素对 `user` 角色可见（不隐藏），`user` 角色同 `admin`。

## 3. 文章列表页（`src/views/articles.ejs`）

```ejs
<%
const isGuest = user && user.role === 'guest';  // 改：仅 guest 为 true
const isGuestValue = isGuest ? 'true' : 'false';
const pageActions = isGuest ? '' : `
    <div class="page-actions">
      <button class="btn btn-secondary" onclick="processBatch()">批量处理</button>
      <button class="btn btn-secondary" onclick="markAllAsRead()">批量已读</button>
    </div>`;
%>
```

文章卡片操作按钮：

```javascript
// user 角色可见操作按钮
${user.role !== 'guest'
  ? `<button class="btn-icon" onclick="toggleReadStatus(...)">...</button>`
  : ''}
${user.role !== 'guest' && (article.process_status === 'pending' || article.process_status === 'failed')
  ? `<button class="btn-icon" onclick="processArticle(...)">处理</button>` : ''}
${user.role !== 'guest'
  ? `<button class="btn-icon" onclick="deleteArticle(...)">删除</button>` : ''}
```

## 4. 页面路由（`src/api/web.ts`）

```typescript
// 所有受限制页面
app.get('/settings', optionalAuth, (req: any, res) => {
  if (!req.userId) return res.redirect('/login');
  if (req.user?.role === 'guest') return res.redirect('/');  // 改：只挡 guest
  res.render('settings', { user: req.user });
});

app.get('/topics', optionalAuth, (req: any, res) => {
  if (!req.userId) return res.redirect('/login');
  if (req.user?.role === 'guest') return res.redirect('/');  // 改：只挡 guest
  res.render('topics', { user: req.user });
});
```

## 5. 客户端角色注入

`window.userRole` 已存在，无需改动。现有 `guestSummaryEnabled` 逻辑不受影响。
